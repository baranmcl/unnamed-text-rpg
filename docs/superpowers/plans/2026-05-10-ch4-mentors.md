# Chapter 4 Mentors Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the Chapter 4 "Meeting with the Mentor" beat — a paced transit zone (Old Road, 3 mandatory combats), four class-specific mentor encounters that grant the class's `signatureMove`, and a hidden achievement for finishing without ever learning it.

**Architecture:** Each class meets a mentor at a class-specific location reached via a new shared transit zone (`the_old_road`). Mentors are previous-cycle heroes who explicitly name "the Sacred MacGuffin." Accepting their teaching grants the class signatureMove (replacing the level-3 auto-unlock). Refusal is reversible (player can return) except for Farmhand, where refusal still grants the skill comedically. The transit zone gates forward exits behind a `old_road_wins >= 3` counter. Two new monsters with a new `plot_armor` status drop a new HP-recovery consumable (`Page of Errata`).

**Tech Stack:** TypeScript (strict), Svelte 5 runes, Vitest. Pure-functional reducer + content registries. All identifiers branded.

**Reference:** `docs/superpowers/specs/2026-05-10-ch4-mentors-design.md`

### Task 3: plot_armor combat hook

The Plot Convenience applies `plot_armor` via its `apply_status` action. By default, `monsterTurn`'s `apply_status` dispatch applies the status to the *player* (see `src/engine/combat.ts:348-357`). For `plot_armor`, we want the monster to buff *itself*. We extend `monsterTurn` so that when `action.status === 'plot_armor'`, the status applies to the monster combatant instead. Then, in the damage-application pipeline, incoming damage to any combatant with `plot_armor` is capped at 1.

- Modify: `src/engine/combat.ts` — `monsterTurn` (line 348-357) for self-targeting; damage pipeline (lines 195-233 inside `playerAttack`) for the cap

```typescript
describe('plot_armor status', () => {
  it('caps incoming damage at 1 while active on a monster', () => {
    let s = createInitialState(123);
    s = reduce(s, { kind: 'StartNewGame', name: 'Tester', classId: 'reluctant_farmhand' as ClassId });
    s = reduce(s, { kind: 'TriggerEncounter', encounterId: 'practice_dummy' as EncounterId });
    if (s.combat?.kind !== 'turn-based') throw new Error('expected combat');
    const monsterId = s.combat.combatants.find((c) => c.kind === 'monster')!.id;

    s = applyStatus(s, { kind: 'combatant', combatantId: monsterId }, {
      kind: 'plot_armor',
      duration: { kind: 'turns', remaining: 2 },
      source: 'test'
    });

    const hpBefore = (s.combat as TurnBasedCombatState).combatants.find((c) => c.kind === 'monster')!.hp;

    // Attack up to 5 times to overcome miss variance — every hit should deal exactly 1.
    let damageDealt = 0;
    for (let i = 0; i < 5; i++) {
      const before = (s.combat as TurnBasedCombatState).combatants.find((c) => c.kind === 'monster')!.hp;
      s = reduce(s, { kind: 'AttackTarget' });
      if (s.combat?.kind !== 'turn-based') break;
      const after = (s.combat as TurnBasedCombatState).combatants.find((c) => c.kind === 'monster')!.hp;
      const dealt = before - after;
      if (dealt > 0) {
        // Each landing hit must be exactly 1 (the cap).
        expect(dealt).toBe(1);
        damageDealt += dealt;
      }
    }
    // We expect to have landed at least one hit and seen the cap apply.
    expect(damageDealt).toBeGreaterThan(0);
    expect(damageDealt).toBeLessThanOrEqual(2); // 2 turns of plot_armor → ≤2 capped hits before it expires
  });
});
```

- (b) Have a "noFlee-flagged" variant for Old Road use

```typescript
  if (s.combat) s = { ...s, combat: null };
  s = { ...s, world: { ...s.world, flags: { ...s.world.flags, accepted_call: true, crossed_threshold: true, old_road_cleared: true, old_road_wins: 3 } } };
  // Jump straight to the teaching node by starting the encounter and advancing.

  s = startNarrativeEncounter(s, enc as import('../../../engine/types').NarrativeEncounter);
  // Advance past arrival → teaching by triggering the first (and only) choice.
```

```typescript
function appendLogs(state: GameState, entries: Omit<LogEntry, 'id'>[]): GameState {
  let nextId = state.log.length === 0 ? 1 : state.log[state.log.length - 1]!.id + 1;
  const withIds: LogEntry[] = entries.map((e) => ({ ...e, id: nextId++ }));
  const merged = [...state.log, ...withIds];
  return { ...state, log: merged.length > MAX_LOG_ENTRIES ? merged.slice(-MAX_LOG_ENTRIES) : merged };
}

function grantSignatureMove(state: GameState): GameState {
  const cls = classes[state.character.classId];
  if (!cls) return state;
  if (state.character.knownSkills.includes(cls.signatureMove)) return state;
  return {
    ...state,
    character: { ...state.character, knownSkills: [...state.character.knownSkills, cls.signatureMove] },
    world: {
      ...state.world,
      flags: {
        ...state.world.flags,
        ever_learned_signature: true,
        'achievements.signature_unlocked': true
      }
    }
  };
}

  // The classId tells us which teaching node to advance to.
  const cls = state.character.classId as string;
  const map: Record<string, string> = {
    disgraced_knight: 'mentor_knight_teaching',
    accidental_wizard: 'mentor_wizard_teaching',
    bard: 'mentor_bard_teaching',
    reluctant_farmhand: 'mentor_farmhand_teaching'
  };
  const nextId = map[cls];
  if (!nextId) return { state, next: null };
  return { state, next: NarrativeNodeId(nextId) };
};
```

```typescript
import { createInitialState } from '../engine/state';
import { ClassId, EncounterId, LocationId, SkillId } from '../engine/types';
import type { GameState, NarrativeCombatState } from '../engine/types';

function setupAtMentorLocationDoor(classId: ClassId, locationId: LocationId, seed = 9): GameState {
  let s = createInitialState(seed);
  s = reduce(s, { kind: 'StartNewGame', name: 'T', classId });
  if (s.combat) s = { ...s, combat: null };
  s = { ...s, world: { ...s.world, flags: { ...s.world.flags, accepted_call: true, crossed_threshold: true, old_road_cleared: true, old_road_wins: 3, [`class.${classId}`]: true } } };
  return s;
}
```
